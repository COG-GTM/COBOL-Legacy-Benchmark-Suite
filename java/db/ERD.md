# Entity-Relationship Diagram — Portfolio Baseline Schema

Relational schema translated from the VSAM copybooks (see `FIELD_MAPPINGS.md` and
`ddl/V1__baseline_schema.sql`).

```mermaid
erDiagram
    PORTFOLIO_MASTER ||--o{ PORTFOLIO_TRANSACTION : "has transactions"
    PORTFOLIO_MASTER ||--o{ PORTFOLIO_POSITION : "has positions"
    PORTFOLIO_MASTER ||--o{ HISTORY_RECORD : "has history"
    PORTFOLIO_MASTER ||..o{ AUDIT_LOG : "referenced by (no FK)"

    PORTFOLIO_MASTER {
        CHAR(8) PORTFOLIO_ID PK "PORT-ID"
        CHAR(10) ACCOUNT_NO PK "PORT-ACCOUNT-NO"
        VARCHAR(30) CLIENT_NAME "PORT-CLIENT-NAME"
        CHAR(1) CLIENT_TYPE "I/C/T"
        DATE CREATE_DATE "PORT-CREATE-DATE"
        DATE LAST_MAINT_DATE "PORT-LAST-MAINT"
        CHAR(1) STATUS "A/C/S"
        NUMERIC(15_2) TOTAL_VALUE "COMP-3"
        NUMERIC(15_2) CASH_BALANCE "COMP-3"
        VARCHAR(8) LAST_MAINT_USER "PORT-LAST-USER"
        BIGINT LAST_TRANS_NO "PORT-LAST-TRANS"
    }

    PORTFOLIO_TRANSACTION {
        DATE TRANS_DATE PK "TRN-DATE"
        TIME TRANS_TIME PK "TRN-TIME"
        CHAR(8) PORTFOLIO_ID PK,FK "TRN-PORTFOLIO-ID"
        CHAR(6) SEQUENCE_NO PK "TRN-SEQUENCE-NO"
        CHAR(10) INVESTMENT_ID "TRN-INVESTMENT-ID"
        CHAR(2) TRANS_TYPE "BU/SL/TR/FE"
        NUMERIC(15_4) QUANTITY "COMP-3"
        NUMERIC(15_4) PRICE "COMP-3"
        NUMERIC(15_2) AMOUNT "COMP-3"
        CHAR(3) CURRENCY_CODE "TRN-CURRENCY"
        CHAR(1) STATUS "P/D/F/R"
        TIMESTAMP PROCESS_DATE "TRN-PROCESS-DATE"
        VARCHAR(8) PROCESS_USER "TRN-PROCESS-USER"
    }

    PORTFOLIO_POSITION {
        CHAR(8) PORTFOLIO_ID PK,FK "POS-PORTFOLIO-ID"
        DATE POSITION_DATE PK "POS-DATE"
        CHAR(10) INVESTMENT_ID PK "POS-INVESTMENT-ID"
        NUMERIC(15_4) QUANTITY "COMP-3"
        NUMERIC(15_2) COST_BASIS "COMP-3"
        NUMERIC(15_2) MARKET_VALUE "COMP-3"
        CHAR(3) CURRENCY_CODE "POS-CURRENCY"
        CHAR(1) STATUS "A/C/P"
        TIMESTAMP LAST_MAINT_DATE "POS-LAST-MAINT-DATE"
        VARCHAR(8) LAST_MAINT_USER "POS-LAST-MAINT-USER"
    }

    HISTORY_RECORD {
        CHAR(8) PORTFOLIO_ID PK,FK "HIST-PORTFOLIO-ID"
        DATE HIST_DATE PK "HIST-DATE"
        TIME HIST_TIME PK "HIST-TIME"
        CHAR(4) SEQ_NO PK "HIST-SEQ-NO"
        CHAR(2) RECORD_TYPE "PT/PS/TR"
        CHAR(1) ACTION_CODE "A/C/D"
        VARCHAR(400) BEFORE_IMAGE "HIST-BEFORE-IMAGE"
        VARCHAR(400) AFTER_IMAGE "HIST-AFTER-IMAGE"
        CHAR(4) REASON_CODE "HIST-REASON-CODE"
        TIMESTAMP PROCESS_DATE "HIST-PROCESS-DATE"
        VARCHAR(8) PROCESS_USER "HIST-PROCESS-USER"
    }

    ERROR_LOG {
        BIGINT ERROR_LOG_ID PK "identity"
        DATE ERROR_DATE "ERR-DATE"
        TIME ERROR_TIME "ERR-TIME"
        VARCHAR(8) PROGRAM_ID "ERR-PROGRAM"
        CHAR(2) ERROR_CATEGORY "VS/VL/PR/SY"
        CHAR(4) ERROR_CODE "ERR-CODE"
        SMALLINT ERROR_SEVERITY "0/4/8/12/16"
        VARCHAR(80) ERROR_TEXT "ERR-TEXT"
        VARCHAR(256) ERROR_DETAILS "ERR-DETAILS"
    }

    AUDIT_LOG {
        BIGINT AUDIT_LOG_ID PK "identity"
        TIMESTAMP AUDIT_TIMESTAMP "AUD-TIMESTAMP"
        VARCHAR(8) SYSTEM_ID "AUD-SYSTEM-ID"
        VARCHAR(8) USER_ID "AUD-USER-ID"
        VARCHAR(8) PROGRAM_ID "AUD-PROGRAM"
        VARCHAR(8) TERMINAL_ID "AUD-TERMINAL"
        CHAR(4) AUDIT_TYPE "TRAN/USER/SYST"
        CHAR(8) AUDIT_ACTION "CREATE..SHUTDOWN"
        CHAR(4) AUDIT_STATUS "SUCC/FAIL/WARN"
        CHAR(8) PORTFOLIO_ID "AUD-PORTFOLIO-ID"
        CHAR(10) ACCOUNT_NO "AUD-ACCOUNT-NO"
        VARCHAR(100) BEFORE_IMAGE "AUD-BEFORE-IMAGE"
        VARCHAR(100) AFTER_IMAGE "AUD-AFTER-IMAGE"
        VARCHAR(100) AUDIT_MESSAGE "AUD-MESSAGE"
    }
```

Notes:
- `PORTFOLIO_MASTER` has a composite VSAM key `(PORTFOLIO_ID, ACCOUNT_NO)`; child tables key on
  `PORTFOLIO_ID` alone (matching the copybook child-record keys), backed by a `UNIQUE` constraint
  on `PORTFOLIO_MASTER.PORTFOLIO_ID`.
- `ERROR_LOG` and `AUDIT_LOG` are insert-only log tables with surrogate identity keys; `AUDIT_LOG`
  references portfolios informally (nullable columns, no FK) because audit rows may record events
  for portfolios that no longer exist.
